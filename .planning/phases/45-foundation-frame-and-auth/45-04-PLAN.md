```typescript
// src/net/bindTable.ts
import type { Ref, ShallowRef } from 'vue';
type Listener = (...args: any[]) => void;
export interface SubscriptionHandleLike { unsubscribe(): void; isActive(): boolean; isEnded(): boolean; }
export interface SubscriptionBuilderLike {
  onApplied(cb: Listener): SubscriptionBuilderLike;
  onError(cb: Listener): SubscriptionBuilderLike;
  subscribe(query: string[]): SubscriptionHandleLike;
}
export interface TableLike<Row> {
  iter(): Iterable<Row>;
  onInsert(cb: Listener): void;  removeOnInsert(cb: Listener): void;
  onDelete(cb: Listener): void;  removeOnDelete(cb: Listener): void;
  onUpdate?(cb: Listener): void; removeOnUpdate?(cb: Listener): void; // views may lack onUpdate
}
export interface ConnLike { subscriptionBuilder(): SubscriptionBuilderLike; }
export interface BindTableOptions<C, Row> { table(conn: C): TableLike<Row>; sql: string[]; filter?(row: Row): boolean; }
export interface TableBinding<C, Row> {
  readonly rows: Readonly<ShallowRef<readonly Row[]>>;
  readonly applied: Readonly<Ref<boolean>>;
  readonly failed: Readonly<Ref<boolean>>;
  attach(conn: C | null): void;   // null keeps rows; same conn is a no-op
  dispose(): void;                // detach + unsubscribe + rows [] + applied false
}
export function bindTable<C extends ConnLike, Row>(options: BindTableOptions<C, Row>): TableBinding<C, Row>;

// src/session/deriveScreen.ts
export type LinkStatus = 'idle' | 'connecting' | 'connected' | 'reconnecting' | 'unreachable' | 'rejected' | 'expired';
// Must equal ConnectionStatus in src/net/connection.ts (45-03, same wave). Keep the seven members identical.
export type SplashState = 'idle' | 'redirecting' | 'connecting' | 'signingIn' | 'sessionExpired' | 'signInFailed' | 'unreachable';
export type AppScreen =
  | { kind: 'splash'; state: SplashState } | { kind: 'picker' } | { kind: 'noCharacters' } | { kind: 'frame' };
export interface ScreenInput {
  redirecting: boolean; authFailed: boolean; hasToken: boolean; status: LinkStatus;
  playerLoaded: boolean;              // my_player applied and a row exists
  userId: bigint | null; activeCharacterId: bigint | null;
  charactersApplied: boolean; characterCount: number; activeCharacterLoaded: boolean;
}
export function deriveScreen(input: ScreenInput): AppScreen;

// src/session/versionCheck.ts
export function shouldPromptReload(row: { version: string } | null | undefined, clientVersion: string | null | undefined, isDev: boolean): boolean;

// src/session/frameView.ts
import type { Character, Location, Region, WorldState, PendingSkill } from '../module_bindings/types';
export type TimeOfDay = 'day' | 'night';
export const UNKNOWN_PLACE = 'Unknown place';
export interface FrameView {
  characterName: string; avatarInitial: string;
  classLine: string;    // 'Lv 6 · Wizard'
  accountLine: string;  // 'Lv 6 · Elf Wizard'
  hp: bigint; maxHp: bigint; mana: bigint; maxMana: bigint; stamina: bigint; maxStamina: bigint;
  placeLabel: string;   // 'Region · Location' or UNKNOWN_PLACE
  locationName: string; // 'Location' or UNKNOWN_PLACE
  timeOfDay: TimeOfDay | null;
  levelUp: boolean; newSkill: boolean;
}
export function describePlace(locationId: bigint | null | undefined, locations: readonly Location[], regions: readonly Region[]): { placeLabel: string; locationName: string };
export function timeOfDay(rows: readonly WorldState[]): TimeOfDay | null;
export function classLine(c: Pick<Character, 'level' | 'className'>): string;
export function accountLine(c: Pick<Character, 'level' | 'race' | 'className'>): string;
export function avatarInitial(name: string): string;               // first character upper-cased, '?' when empty
export function sortCharacters(chars: readonly Character[]): Character[]; // createdAt.microsSinceUnixEpoch ascending, tie by id
export function buildFrameView(input: { character: Character; locations: readonly Location[]; regions: readonly Region[]; worldState: readonly WorldState[]; pendingSkills: readonly PendingSkill[] }): FrameView;
```

Location: `id, name, regionId`. Region: `id, name`. WorldState: `isNight`. PendingSkill: `characterId`. AppVersion: `version`.
