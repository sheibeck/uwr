```typescript
// src/net/connection.ts
import type { Ref, ShallowRef } from 'vue';
export type ConnectionStatus =
  'idle' | 'connecting' | 'connected' | 'reconnecting' | 'unreachable' | 'rejected' | 'expired';
// NOTE: 45-04's deriveScreen declares LinkStatus with these exact seven members; keep them identical.
export interface ConnectionHandlers<C> {
  onConnect(conn: C): void;
  onDisconnect(err?: Error): void;
  onConnectError(err: Error): void;
}
export interface ControllerDeps<C extends { disconnect(): void }> {
  build(token: string, handlers: ConnectionHandlers<C>): C;
  getToken(): string | null;       // getStoredIdToken
  hasExpiredToken(): boolean;
  clearSession(): void;            // clearAuthSession
  probe(): Promise<boolean>;       // true when the host answers /v1/ping
  windowTarget?: Pick<EventTarget, 'addEventListener' | 'removeEventListener'> | null;   // 'online'
  documentTarget?: Pick<EventTarget, 'addEventListener' | 'removeEventListener'> | null; // 'visibilitychange'
  isVisible?: () => boolean;
}
export interface ConnectionController<C> {
  readonly status: Readonly<Ref<ConnectionStatus>>;
  readonly conn: Readonly<ShallowRef<C | null>>;   // last connection that reached onConnect; kept (stale) during a retry
  readonly nextRetryAt: Readonly<Ref<number | null>>; // epoch ms of the scheduled retry, null when none
  connect(): void;
  disconnect(): void;   // intentional: no retry, status 'idle', conn null
  retryNow(): void;     // only acts while status is 'reconnecting' or 'unreachable'
  dispose(): void;      // disconnect + remove resume listeners
}
export const MAX_TOKEN_FAILURES: 3;
export function createConnectionController<C extends { disconnect(): void }>(deps: ControllerDeps<C>): ConnectionController<C>;
export function toHttpBase(wsHost: string): string; // ws:// -> http://, wss:// -> https://, trailing slash removed
export function buildDbConnection(token: string, handlers: ConnectionHandlers<DbConnection>): DbConnection;
export function defaultControllerDeps(): ControllerDeps<DbConnection>;
```
