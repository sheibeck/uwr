```typescript
// src/frame/HeaderBar.vue
defineProps<{ placeLabel: string; timeOfDay: 'day' | 'night' | null; levelUp: boolean; newSkill: boolean;
  activeScreen: ScreenId | null; characterName: string; accountLine: string; disabled?: boolean }>();
defineEmits<{ 'toggle-screen': [screen: ScreenId, opener: HTMLElement]; logout: [] }>();

// src/frame/AccountMenu.vue
defineProps<{ characterName: string; accountLine: string }>();
defineEmits<{ logout: [] }>();

// src/frame/tabs.ts
export type TabId = 'story' | 'map' | 'bag' | 'party' | 'more';
export const TABS: readonly { id: TabId; label: string; icon: Component }[]; // Story PhScroll, Map PhMapTrifold, Bag PhBackpack, Party PhUsersThree, More PhDotsThree
export function screenForTab(tab: TabId): ScreenId | 'more' | null; // story->null, map->'map', bag->'bag', party->'social', more->'more'
export function tabForScreen(active: ScreenId | 'more' | null): TabId; // null->story, map->map, bag->bag, social->party, anything else->more

// src/frame/TabBar.vue
defineProps<{ activeTab: TabId; sheetOpen: boolean }>();
defineEmits<{ select: [tab: TabId, opener: HTMLElement] }>();

// src/frame/LocationRow.vue
defineProps<{ locationName: string; timeOfDay: 'day' | 'night' | null }>();

// src/frame/NoticeBars.vue
defineProps<{ reconnecting: boolean; nextRetryAt: number | null; versionPrompt: boolean }>();
defineEmits<{ reload: [] }>();
```
