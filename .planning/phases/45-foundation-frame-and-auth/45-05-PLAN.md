Build the screen-state machine and the two overlay shells every secondary screen opens in: a desktop drawer and a mobile full-height sheet, plus the More sheet. Handle one-at-a-time, Esc, close button, focus to the close button on open, focus back to the opener on close, and a Tab-wrap trap that leaves the header and rails clickable.

Output: `useBreakpoint.ts`, `useScreens.ts`, `focusTrap.ts`, `Drawer.vue`, `Sheet.vue`, `MoreSheet.vue` and their tests in `src/frame/`.
</objective>

```typescript
export const DESKTOP_QUERY = '(min-width: 900px)';
export function useBreakpoint(): { isDesktop: Readonly<Ref<boolean>> }; // matchMedia + 'change' listener, removed on scope dispose

// src/frame/useScreens.ts
export type ActiveScreen = ScreenId | 'more' | null;
export interface ScreensApi {
  readonly active: Readonly<Ref<ActiveScreen>>;
  open(screen: ScreenId | 'more', opener: HTMLElement | null): void; // replaces any open screen
  toggle(screen: ScreenId, opener: HTMLElement | null): void;        // desktop header: close when already open
  openFromMore(screen: ScreenId): void;                              // keeps the More tab as opener
  close(): void;                                                     // active null, focus back to opener after nextTick
  syncLayout(isDesktop: boolean): void;                              // 'more' becomes null on desktop
}
export function useScreens(): ScreensApi;

// src/frame/focusTrap.ts
export const FOCUSABLE_SELECTOR: string;
export function focusableWithin(container: HTMLElement): HTMLElement[];
export function trapTabKey(event: KeyboardEvent, container: HTMLElement): void;

// src/frame/Drawer.vue and src/frame/Sheet.vue
defineProps<{ title: string }>(); defineEmits<{ close: [] }>(); // default slot = body
// src/frame/MoreSheet.vue
defineEmits<{ select: [screen: ScreenId]; logout: []; close: [] }>();
```

Design rules for every component in this plan (enforced by 45-02's guards): colors only via var(--…), transparent, currentColor or color-mix(in srgb, var(--…) N%, transparent); font sizes 10/12/14/20 px; weights 400/500; padding, margin and gap only 0/4/8/16/24/32/48/64 px; no font-family; Phosphor components only (no svg markup); headings only h4/h6; no v-html; numbers use font-variant-numeric: tabular-nums.
