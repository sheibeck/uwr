Layer the client tokens and the frame-level overrides on top of the vendored Nocturne stylesheet, and add the static guards that enforce the design contract for every later plan: no literal colors, pinned hues, defined variables, the four-size type scale, two weights, the 4-multiple spacing scale, Inter only, Phosphor only.

Output: `src/styles/tokens.client.css`, `src/styles/frame.css`, `src/styles/cssContract.ts`, three contract tests, two import lines in `src/main.ts`.

```typescript
// src/styles/cssContract.ts  (node:fs + postcss + vue/compiler-sfc; not imported by app code)
export interface CssDecl { file: string; selector: string | null; atRule: string | null; prop: string; value: string; }
export interface ClientFiles { css: string[]; vue: string[]; ts: string[]; } // absolute paths, forward slashes
export const NAMED_COLORS: ReadonlySet<string>;            // the 148 CSS named colors, lower case
export function listClientFiles(srcRoot: string): ClientFiles; // walks src/, skips module_bindings and *.test.ts
export function sfcStyleBlocks(source: string): string[];     // parse(source).descriptor.styles[i].content
export function sfcTemplateAndScript(source: string): string; // template + script + scriptSetup content
export function parseDecls(css: string, file: string): CssDecl[]; // postcss walkDecls; selector of parent rule; params of enclosing @media/@container/@supports
export function colorOffenders(css: string, file: string): string[]; // literal colors in color-bearing properties
export function textColorOffenders(text: string, file: string): string[]; // hex + functional color notations in TS/template text
export function definedCustomProperties(css: string): Set<string>;   // '--name' declarations
export function usedCustomProperties(text: string): Set<string>;     // names inside var(--name ...)
```
