// Pure module. Allowed imports: truncateCodePoints from '../data/llm_layers' only.
export const SEGMENT_KINDS = ['narration', 'dialogue'] as const;
export type SegmentKind = (typeof SEGMENT_KINDS)[number];
export const KEEPER_SPEAKER = 'The Keeper';
export const MAX_SEGMENTS = 6;
export const MAX_SEGMENT_CHARS = 600;        // Unicode code points
export const TRUNCATION_MARK = '…';
export type Segment = { kind: SegmentKind; speaker: string; text: string; speakerNpcId?: bigint };
export type PresentSpeaker = { name: string; id?: bigint };
export type SegmentSource = 'segments' | 'legacy_dialogue' | 'legacy_narrative' | 'prose' | 'fallback';
export interface ReplyLadderOptions {
  present: readonly PresentSpeaker[];        // who may speak in a dialogue segment
  playerNames: readonly string[];            // dialogue attributed to these (or to "you") is dropped
  fallbackLine: string;                      // in-voice line when nothing usable remains
  legacyDialogueSpeaker?: PresentSpeaker;    // npc_conversation: salvage a top-level "dialogue" string (OQ2)
  legacyNarrativeField?: boolean;            // combat_narration: salvage a top-level "narrative" string
  salvageProse?: boolean;                    // combat_narration: wrap usable non-JSON prose as Keeper narration
  cleanProse?: (text: string) => string;     // combat_narration passes stripNarrationSelfCorrection
}
export interface ReplyLadderResult { segments: Segment[]; source: SegmentSource; parsed: Record<string, unknown> | undefined }
export function speakerKey(name: string): string;
export function cleanSegmentText(raw: unknown, kind: SegmentKind): string;
export function normalizeSegments(raw: unknown, present: readonly PresentSpeaker[], playerNames?: readonly string[]): Segment[];
export function keeperSegments(text: string): Segment[];
export function keeperFallback(line: string): Segment[];
export function flattenSegments(segments: readonly Segment[]): string;
export function parseReplyObject(raw: unknown): Record<string, unknown> | undefined;
export function isUsableProse(text: string): boolean;
export function segmentsFromReply(resultText: unknown, options: ReplyLadderOptions): ReplyLadderResult;

<assumption_delta_decision>
Primary noun: `segments`, typed `KeeperSegment[]` in the schema and `Segment[]` in module code, is the canonical representation of a narrative reply.
Decision: PROMOTE. Segments are primary. `message` is DERIVED from them with `flattenSegments(segments)` and is kept only for logs, admin views and consumers that ignore segments. No code path may write a narrative event whose `message` disagrees with its `segments`.
Where it binds: this plan provides `flattenSegments` and the storage; 46-02 and 46-03 write every narrative row as `message = flattenSegments(segs)` with `segs` passed alongside; 46-03 adds the companion invariant test (for every narrative route's stored event, `message === flattenSegments(segments)`).
Rows without segments (system lines, effect cues, quest rows, all callers outside the narrative routes) keep `message` as their only representation; that is unchanged.
</assumption_delta_decision>
