```typescript
// src/input/commands.ts
export const COMMAND_WORDS = ['who','accept','decline','leave','invite','kick','promote','whisper','w',
  'friend','endcombat','end','endc','group','renown','factions','faction','events'] as const;
export type CommandWord = (typeof COMMAND_WORDS)[number];
export type CommandShape = 'bare' | 'name' | 'nameAndMessage';
export const COMMAND_SHAPES: Readonly<Record<CommandWord, CommandShape>>;
export function tokenize(text: string): string[];            // trim, split on /\s+/, '' -> []
export function isCommandWord(token: string): token is CommandWord; // exact match after toLowerCase()

// src/input/conversation.ts
export const FAREWELL_WORDS: readonly string[];   // bye farewell leave goodbye end quit exit back
export const BARE_GAME_ACTIONS: readonly string[];
export function isFarewell(text: string): boolean;
export function isGameAction(text: string, ctx: { placeNames: readonly string[]; nodeNames: readonly string[] }): boolean;

// src/input/routeInput.ts
export interface NamedId { id: bigint; name: string }
export interface RouteContext {
  conversation: NamedId | null;          // NPC currently in conversation
  npcsHere: readonly NamedId[];
  placeNames: readonly string[];         // connected location names
  nodeNames: readonly string[];          // resource node names here
  pendingInviterNames: readonly string[];
}
export type ReducerCall =
  | { reducer: 'whisper'; args: { targetName: string; message: string } }
  | { reducer: 'groupMessage'; args: { message: string } }
  | { reducer: 'inviteToGroup' | 'kickGroupMember' | 'promoteGroupLeader' | 'sendFriendRequestToCharacter'; args: { targetName: string } }
  | { reducer: 'acceptGroupInvite' | 'rejectGroupInvite'; args: { fromName: string } }
  | { reducer: 'leaveGroup' | 'endCombat'; args: Record<string, never> }
  | { reducer: 'submitCommand'; args: { text: string } };
export type InfoCommand = 'renown' | 'factions' | 'faction' | 'events' | 'group';
export type Route =
  | { kind: 'none' }
  | { kind: 'reducer'; call: ReducerCall; echo: string | null }
  | { kind: 'intent'; text: string; echo: string | null; queue: boolean; endsConversation: boolean }
  | { kind: 'talk'; npcId: bigint; message: string }
  | { kind: 'hail'; npc: NamedId; text: string; echo: string }
  | { kind: 'info'; command: InfoCommand; arg: string | null; echo: string }
  | { kind: 'endConversation'; echo: string };
export function routeInput(text: string, ctx: RouteContext): Route;

// src/input/history.ts
export const HISTORY_LIMIT = 50;
export interface InputHistory {
  record(text: string): void;        // also resets the recall cursor
  up(draft: string): string | null;  // null when there is nothing older
  down(): string | null;             // null when not recalling
  resetCursor(): void;
  entries(): readonly string[];      // newest first
}
export function createInputHistory(limit?: number): InputHistory;
```
`echo` holds the trimmed typed text WITHOUT the "› " prefix (the feed renderer adds it, 47-07). `args` never includes `characterId`; 47-09 adds it.

Reducer argument names come from the generated bindings (src/module_bindings/*_reducer.ts): whisper {targetName, message}; group_message {message}; invite_to_group / kick_group_member / promote_group_leader / send_friend_request_to_character {targetName}; accept_group_invite / reject_group_invite {fromName}; leave_group / end_combat {}; submit_command {text}.
