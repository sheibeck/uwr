// src/game/bindEventTable.ts
import type { ConnLike } from '../net/bindTable';
type Listener = (...args: any[]) => void;
export interface EventTableLike<Row> { onInsert(cb: Listener): void; removeOnInsert(cb: Listener): void }
export interface BindEventTableOptions<C, Row> { table(conn: C): EventTableLike<Row>; sql: string[]; onRow(row: Row): void }
export interface EventTableBinding<C> {
  readonly applied: Readonly<Ref<boolean>>;
  readonly failed: Readonly<Ref<boolean>>;
  attach(conn: C | null): void;   // same conn is a no-op; null detaches
  dispose(): void;
}
export function bindEventTable<C extends ConnLike, Row>(options: BindEventTableOptions<C, Row>): EventTableBinding<C>;

// src/game/keyedBinding.ts
export interface AttachableBinding<C> { readonly applied: Readonly<Ref<boolean>>; attach(conn: C | null): void; dispose(): void }
export interface Keyed<B> {
  readonly current: Readonly<ShallowRef<B | null>>;   // the binding whose data is shown
  reset(): void;                                      // dispose current and pending, forget the key
}
export function createKeyed<C, K extends string | bigint, B extends AttachableBinding<C>>(options: {
  key: Readonly<Ref<K | null>>;
  conn: Readonly<Ref<C | null>>;
  make(key: K): B;
  swap?: 'onApplied' | 'immediate';   // default 'onApplied'; event bindings use 'immediate'
}): Keyed<B>;
export function keyedRows<Row>(keyed: Keyed<{ rows: Readonly<Ref<readonly Row[]>> } & AttachableBinding<any>>): ComputedRef<readonly Row[]>;
export function idListKey(ids: Iterable<bigint>): string | null;   // '5,9' (sorted numerically, deduped); null when empty
export function parseIdListKey(key: string): bigint[];

// src/game/serverClock.ts

    - key 'a' -> 'b' with swap 'onApplied': make('b') is attached as pending; current stays 'a' (its rows still shown) until b.applied becomes true; then 'a' is disposed and 'b' becomes current
    - key 'b' -> 'c' while 'b' is still pending: pending 'b' is disposed, 'c' becomes pending, 'a' stays current
    - swap 'immediate': 'a' -> 'b' disposes 'a' and makes 'b' current at once
    - key -> null disposes current and pending; current null

    - reset(): disposes current and pending; a later key change (null -> 'a') makes a new binding
    - the same key value set again (equal string or bigint) does not rebind
