export const QUEUE_MAX = 3;
export interface QueuedLine { text: string; mode: 'narrative' | 'intent'; echoKey: string }
export interface NarrativeQueue {
  readonly items: Readonly<ShallowRef<readonly QueuedLine[]>>;
  readonly inFlight: Readonly<Ref<boolean>>;
  mustQueue(gateActive: boolean): boolean;   // gate || inFlight || items non-empty
  enqueue(line: QueuedLine): boolean;        // false when full
  takeNext(gateActive: boolean): QueuedLine | null;  // head when !gate && !inFlight; marks in flight
  beginDirect(): void;                       // a direct (unqueued) narrative send is in flight
  settle(): void;                            // the in-flight send finished (resolved or rejected)
  drop(): QueuedLine[];                      // empty the queue, clear inFlight, return dropped lines
}
export function createNarrativeQueue(max?: number): NarrativeQueue;

// src/input/infoCommands.ts
export function standingLabel(standing: bigint): string;   // 'Hated' .. 'Exalted'

export function formatFactions(input: { standings: readonly { factionId: bigint; standing: bigint }[]; factions: readonly { id: bigint; name: string }[] }): string;
export function formatFaction(name: string, input: { standings: readonly { factionId: bigint; standing: bigint }[]; factions: readonly { id: bigint; name: string; description: string }[] }): string;

  regions: readonly { id: bigint; name: string }[]; objectives: readonly { eventId: bigint; name: string; currentCount: bigint; targetCount: bigint }[] }): string;

  members: readonly { id: bigint; name: string; level: bigint; className: string }[]; inviterNames: readonly string[] }): string;

// src/console/feedStore.ts
export const FEED_LINE_CAP = 300;
export type FeedSource = 'private' | 'location' | 'group' | 'world' | 'local';
export type LocalKind = 'echo' | 'system' | 'look';
export interface SegmentRowLike { kind: string; speaker: string; text: string; speakerNpcId?: bigint | null }
export interface EventRowLike { id: bigint; kind: string; message: string; createdAt: { microsSinceUnixEpoch: bigint };
  segments?: readonly SegmentRowLike[] | null; characterId?: bigint; excludeCharacterId?: bigint | null }
export interface FeedEntry { key: string; source: FeedSource; id: bigint; kind: string; message: string;
  segments: readonly SegmentRowLike[] | null; characterId: bigint | null; createdAtMicros: bigint; queued: boolean; lineCount: number }
export interface FeedStore {
  readonly entries: Readonly<ShallowRef<readonly FeedEntry[]>>;
  readonly characterId: Readonly<Ref<bigint | null>>;
  setCharacter(id: bigint | null): void;     // clears history and the pending batch when the id changes
  ingest(source: Exclude<FeedSource, 'local'>, row: EventRowLike): void;   // buffered; flushed in a microtask
  appendLocal(kind: LocalKind, message: string, options?: { queued?: boolean }): string;  // returns the key
  setQueued(key: string, queued: boolean): void;
  remove(key: string): void;
  clear(): void;
  flush(): void;                              // flush the pending batch now
}
export function acceptRow(source: Exclude<FeedSource, 'local'>, row: EventRowLike, characterId: bigint | null): boolean;
export function createFeedStore(options?: { cap?: number; schedule?: (fn: () => void) => void }): FeedStore;
